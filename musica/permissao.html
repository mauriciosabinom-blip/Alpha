Refrão · liberar o servidor das notificações

  1. Toque em Copiar aqui embaixo.
  2. Abra o SQL Editor do Supabase.
  3. Toque no editor, cole e toque em Run.
  4. Em Results deve aparecer servidor_pode_ler = true.
grant usage on schema public to service_role;
grant select, insert, update, delete on public.app_secrets, public.push_subscriptions, public.profiles, public.messages, public.friendships, public.karaoke_scores, public.login_attempts, public.blocked_words to service_role;
grant usage, select on all sequences in schema public to service_role;
select has_table_privilege('service_role', 'public.app_secrets', 'select') as servidor_pode_ler;
